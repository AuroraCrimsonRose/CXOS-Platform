# Kernel code, compiled for the HOST and tested under AddressSanitizer and
# UndefinedBehaviorSanitizer (docs/safety/CX_SAFETY.md section 3.7).
#
# A separate CMake project from tools/cmake: that one cross-compiles the
# kernel for i686-elf, this one builds an ordinary host executable from the
# same sources. Only code with no hardware dependency belongs in KERNEL_SOURCES;
# anything that touches ports, page tables or interrupts is tested by booting.
#
#   cmake -S kernel/test/host -B build/hosttest -G Ninja -DCMAKE_C_COMPILER=clang
#   ninja -C build/hosttest && build/hosttest/khost_tests
#
# CXEX.Tests' KernelHostTests runs exactly that as part of `dotnet test`.
# On Windows the sanitizer runtime is a DLL in clang's resource directory
# (lib/windows); that test puts it on PATH for the run.

cmake_minimum_required(VERSION 3.20)
project(cxk_host_tests C)

set(KDIR ${CMAKE_CURRENT_SOURCE_DIR}/../..)

set(KERNEL_SOURCES
    ${KDIR}/lib/mem/bitmap.c
    ${KDIR}/lib/crypto/sha256.c
    ${KDIR}/lib/crypto/chacha20.c
    ${KDIR}/lib/crypto/fortuna.c
    ${KDIR}/object/kobj.c
    ${KDIR}/object/htab.c
    ${KDIR}/object/idtab.c
    ${KDIR}/memman/ocache.c
    ${KDIR}/memman/buddy.c
    ${KDIR}/memman/vmtree.c
    ${KDIR}/arch/x86/cpufeat.c
    ${KDIR}/arch/x86/madt.c
    ${KDIR}/arch/x86_64/pagetable.c
    ${KDIR}/drivers/iowin.c
    ${KDIR}/platform/pc/ps2kbd.c
    ${KDIR}/lib/math/kmath.c
    ${KDIR}/diag/crashrec.c
    ${KDIR}/diag/hwfmt.c
    ${KDIR}/diag/panicrep.c
    ${KDIR}/diag/ksym.c
    ${KDIR}/../vendor/qrcodegen/qrcodegen.c
    ${KDIR}/sync/lockorder.c
    ${KDIR}/drivers/storage/filesys/cxfs.c
    ${KDIR}/lib/time/datetime.c
)

# The seam CXFS runs on (hostdisk.c: RAM disks behind the registry's interface,
# the storage lock, uid and clock), the kernel's strlcpy, and the one-input
# driver the fuzz target and test_cxfs.c share.
set(CXFS_HARNESS
    ${CMAKE_CURRENT_SOURCE_DIR}/hostdisk.c
    ${CMAKE_CURRENT_SOURCE_DIR}/kstring.c
    ${CMAKE_CURRENT_SOURCE_DIR}/cxfs_one.c
)

# cxfs.c includes its neighbours by bare name ("disk.h", "string.h"), as the
# kernel build's include list allows. -iquote gives it the kernel's headers
# for those without letting them shadow the host's <string.h> anywhere else.
set(CXFS_QUOTE_DIRS abi kernel/lib/time kernel/lib/string kernel/cpu
    kernel/drivers/storage kernel/drivers/storage/hardware kernel/drivers/storage/filesys)
set(CXFS_QUOTE)
foreach(d ${CXFS_QUOTE_DIRS})
    list(APPEND CXFS_QUOTE "-iquote${KDIR}/../${d}")
endforeach()
set_source_files_properties(
    ${KDIR}/drivers/storage/filesys/cxfs.c ${CMAKE_CURRENT_SOURCE_DIR}/hostdisk.c
    PROPERTIES COMPILE_OPTIONS "${CXFS_QUOTE}")

# The CXFS corpus (corpus/cxfs: seeds from real volumes, and what the fuzzer
# kept) and the inputs it crashed on (regress/cxfs), listed for test_cxfs.c,
# which replays every one through cxfs_one.
file(GLOB CXFS_CORPUS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/corpus/cxfs/*)
file(GLOB CXFS_REGRESS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/regress/cxfs/*)
set(CXFS_LIST "/* generated by kernel/test/host/CMakeLists.txt */
")
foreach(f ${CXFS_REGRESS} ${CXFS_CORPUS})
    string(APPEND CXFS_LIST "\"${f}\",
")
endforeach()
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/cxfs_inputs.h CONTENT "${CXFS_LIST}")
list(LENGTH CXFS_REGRESS CXFS_NREGRESS)

# Vendored and unmodified, so its warnings are not ours to fix.
set_source_files_properties(${KDIR}/../vendor/qrcodegen/qrcodegen.c PROPERTIES COMPILE_OPTIONS "-w")

file(GLOB TEST_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/test_*.c)

add_executable(khost_tests ${CMAKE_CURRENT_SOURCE_DIR}/runner.c ${TEST_SOURCES} ${KERNEL_SOURCES} ${CXFS_HARNESS})

target_include_directories(khost_tests PRIVATE ${KDIR} ${KDIR}/../abi ${KDIR}/../vendor/qrcodegen ${CMAKE_CURRENT_BINARY_DIR})

# The fixed QR record, for the DevKit's real reader (PanicScreenTests).
add_executable(panic_qr_dump ${CMAKE_CURRENT_SOURCE_DIR}/panic_qr_dump.c
               ${KDIR}/diag/panicrep.c ${KDIR}/../vendor/qrcodegen/qrcodegen.c)
target_include_directories(panic_qr_dump PRIVATE ${KDIR} ${KDIR}/../vendor/qrcodegen)
target_compile_definitions(panic_qr_dump PRIVATE CXK_HOST_TEST)
set_property(TARGET panic_qr_dump PROPERTY C_STANDARD 11)
if(WIN32)
    target_compile_definitions(panic_qr_dump PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
# A 2 MB window instead of the kernel's 64 MB, so a test can fill it.
target_compile_definitions(khost_tests PRIVATE CXK_HOST_TEST OCACHE_SLOTS=64
    CXFS_NREGRESS=${CXFS_NREGRESS})
set_property(TARGET khost_tests PROPERTY C_STANDARD 11)

target_compile_options(khost_tests PRIVATE
    -g -O1 -Wall -Wextra -Werror
    -fno-omit-frame-pointer
    -fsanitize=address,undefined
    -fno-sanitize-recover=all
)
target_link_options(khost_tests PRIVATE -fsanitize=address,undefined)

if(WIN32)
    # clang targeting MSVC warns about the CRT's own declarations otherwise.
    target_compile_definitions(khost_tests PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    # test_kmath.c checks lib/math/kmath.c against the host's libm; on an ELF
    # host that is a separate library (the CRT carries it on Windows).
    target_link_libraries(khost_tests PRIVATE m)
endif()

# ---- the CXFS fuzz target (CX_SAFETY.md 3.2) ----
#
# libFuzzer, AddressSanitizer and UndefinedBehaviorSanitizer over CXFS alone
# (fuzz_cxfs.c, cxfs_one.h). Built with everything else when the toolchain can
# link a libFuzzer program; when it cannot, cxfs_fuzz_missing.txt says why and
# KernelHostTests skips its run with that reason.
include(CheckCSourceCompiles)
set(CMAKE_REQUIRED_FLAGS "-fsanitize=fuzzer,address,undefined")
check_c_source_compiles("
#include <stdint.h>
#include <stddef.h>
int LLVMFuzzerTestOneInput(const uint8_t *d, size_t n) { (void)d; (void)n; return 0; }
" CXK_HAVE_LIBFUZZER)
unset(CMAKE_REQUIRED_FLAGS)

if(CXK_HAVE_LIBFUZZER)
    file(REMOVE ${CMAKE_CURRENT_BINARY_DIR}/cxfs_fuzz_missing.txt)
    add_executable(cxfs_fuzz ${CMAKE_CURRENT_SOURCE_DIR}/fuzz_cxfs.c
        ${KDIR}/drivers/storage/filesys/cxfs.c ${KDIR}/lib/time/datetime.c ${CXFS_HARNESS})
    target_include_directories(cxfs_fuzz PRIVATE ${KDIR} ${KDIR}/../abi)
    target_compile_definitions(cxfs_fuzz PRIVATE CXK_HOST_TEST)
    set_property(TARGET cxfs_fuzz PROPERTY C_STANDARD 11)
    target_compile_options(cxfs_fuzz PRIVATE
        -g -O1 -Wall -Wextra -Werror -fno-omit-frame-pointer
        -fsanitize=fuzzer,address,undefined -fno-sanitize-recover=all)
    target_link_options(cxfs_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
    if(WIN32)
        target_compile_definitions(cxfs_fuzz PRIVATE _CRT_SECURE_NO_WARNINGS)
        # libFuzzer on Windows is built against the static CRT; one CRT per program.
        set_property(TARGET cxfs_fuzz PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
    endif()
else()
    file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/cxfs_fuzz_missing.txt
        "${CMAKE_C_COMPILER} could not link a program with -fsanitize=fuzzer,address,undefined
")
endif()
