# Memory
megs: 2048

# CPU - ips (instructions per second) paces Bochs's emulated time against
# real wall-clock time.
# IMPORTANT: ips is a SIGNED 32-bit value - the max is 2147483647 (~2.1 billion).
# Setting it higher overflows to a negative value and breaks timing entirely
# (the log will show a negative "IPS is set to ..." line). Stay at/below ~2.1B.
# On this host even the max is still slightly fast; that is a Bochs limitation.
# For accurate wall-clock time later, use the RTC, not the PIT. QEMU keeps
# better time if you need it now.
cpu: ips=2100000000

# Clock - sync emulated time to the real host clock.
#   sync=realtime  -> Bochs throttles so emulated time tracks wall-clock,
#                     instead of depending on a perfect ips value (which we
#                     can't reach anyway due to the 32-bit ips cap).
#   time0=local    -> initialize the CMOS/RTC to the host's local time, so
#                     the emulated machine's real-time clock is accurate.
clock: sync=realtime, time0=local
